cmake_minimum_required(VERSION 3.20)
project(mpv C CXX)

# ---------------------------------------------------------------------------
# This CMakeLists.txt is a thin wrapper around the real Meson build system.
# It exists so IDEs (CLion, VS Code CMake Tools, etc.) can open the project.
#
# Build directory: <source_root>/buildDir
# To configure manually:  meson setup buildDir
# To build manually:       meson compile -C buildDir
# ---------------------------------------------------------------------------

set(MESON_BUILD_DIR "${CMAKE_SOURCE_DIR}/buildDir" CACHE PATH "Meson build directory")

# ---------------------------------------------------------------------------
# Re-use compile_commands.json produced by Meson so IDE indexing works.
# ---------------------------------------------------------------------------
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(EXISTS "${MESON_BUILD_DIR}/compile_commands.json")
    # Symlink Meson's compile_commands.json into the CMake build dir so IDEs
    # that look for it in CMAKE_BINARY_DIR also find it.
    execute_process(
        COMMAND ${CMAKE_COMMAND} -E create_symlink
            "${MESON_BUILD_DIR}/compile_commands.json"
            "${CMAKE_BINARY_DIR}/compile_commands.json"
        ERROR_QUIET
    )
endif()

# ---------------------------------------------------------------------------
# Meson setup target  (run once, or re-run to reconfigure)
# ---------------------------------------------------------------------------
find_program(MESON_EXECUTABLE meson REQUIRED)
find_program(NINJA_EXECUTABLE ninja)

if(NINJA_EXECUTABLE)
    set(MESON_BACKEND ninja)
else()
    set(MESON_BACKEND auto)
endif()

add_custom_target(meson_setup
    COMMAND ${MESON_EXECUTABLE} setup
        --backend=${MESON_BACKEND}
        --buildtype=debugoptimized
        "${MESON_BUILD_DIR}"
        "${CMAKE_SOURCE_DIR}"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    COMMENT "Running: meson setup ${MESON_BUILD_DIR}"
    VERBATIM
)

# ---------------------------------------------------------------------------
# Main build target — delegates to meson compile
# ---------------------------------------------------------------------------
add_custom_target(meson_build ALL
    COMMAND ${MESON_EXECUTABLE} compile -C "${MESON_BUILD_DIR}"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    COMMENT "Running: meson compile -C buildDir"
    VERBATIM
)

# ---------------------------------------------------------------------------
# Convenience targets
# ---------------------------------------------------------------------------
add_custom_target(meson_test
    COMMAND ${MESON_EXECUTABLE} test -C "${MESON_BUILD_DIR}"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    COMMENT "Running: meson test -C buildDir"
    VERBATIM
)

add_custom_target(meson_install
    COMMAND ${MESON_EXECUTABLE} install -C "${MESON_BUILD_DIR}"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    COMMENT "Running: meson install -C buildDir"
    VERBATIM
)

add_custom_target(meson_clean
    COMMAND ${NINJA_EXECUTABLE} -C "${MESON_BUILD_DIR}" -t clean
    WORKING_DIRECTORY "${MESON_BUILD_DIR}"
    COMMENT "Cleaning Meson build directory"
    VERBATIM
)

add_custom_target(meson_run
    COMMAND "${MESON_BUILD_DIR}/mpv" ${MPV_ARGS}
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    COMMENT "Running mpv"
    VERBATIM
)

# ---------------------------------------------------------------------------
# Dummy library so CLion sees ALL source files and enables navigation/refactor
# (compile flags come from compile_commands.json, not from here)
# ---------------------------------------------------------------------------
file(GLOB_RECURSE ALL_SOURCES
    "${CMAKE_SOURCE_DIR}/audio/*.c"
    "${CMAKE_SOURCE_DIR}/common/*.c"
    "${CMAKE_SOURCE_DIR}/demux/*.c"
    "${CMAKE_SOURCE_DIR}/filters/*.c"
    "${CMAKE_SOURCE_DIR}/input/*.c"
    "${CMAKE_SOURCE_DIR}/misc/*.c"
    "${CMAKE_SOURCE_DIR}/options/*.c"
    "${CMAKE_SOURCE_DIR}/osdep/*.c"
    "${CMAKE_SOURCE_DIR}/player/*.c"
    "${CMAKE_SOURCE_DIR}/stream/*.c"
    "${CMAKE_SOURCE_DIR}/sub/*.c"
    "${CMAKE_SOURCE_DIR}/ta/*.c"
    "${CMAKE_SOURCE_DIR}/video/*.c"
)

add_library(mpv_sources OBJECT EXCLUDE_FROM_ALL ${ALL_SOURCES})

# Point to the config.h generated by Meson so the IDE resolves macros
target_include_directories(mpv_sources PUBLIC
    "${CMAKE_SOURCE_DIR}"
    "${MESON_BUILD_DIR}"
)

message(STATUS "")
message(STATUS "mpv — Meson wrapper CMakeLists.txt")
message(STATUS "  Meson build dir : ${MESON_BUILD_DIR}")
message(STATUS "  To build        : cmake --build . --target meson_build")
message(STATUS "  To reconfigure  : cmake --build . --target meson_setup")
message(STATUS "")
